            if (!isTeacherMode) {
                const password = prompt("Lehrer-Passwort eingeben:");
                if (password === "lehrer123") {
                    setTeacherMode(true);
                } else {
                    alert("Falsches Passwort!");
                }
            } else {
                setTeacherMode(false);
            }
        });

        function setTeacherMode(enabled) {
            isTeacherMode = enabled;
            if (enabled) {
                onboarding.classList.add('hidden');
                studentDashboard.classList.add('hidden');
                teacherDashboard.classList.remove('hidden');
                viewToggle.innerHTML = `<span class="material-symbols-outlined text-sm">person</span> Schüler-Ansicht`;
                viewToggle.classList.replace('bg-primary', 'bg-surface-container-high');
                viewToggle.classList.replace('text-on-primary', 'text-on-surface');
            } else {
                teacherDashboard.classList.add('hidden');
                onboarding.classList.remove('hidden');
                viewToggle.innerHTML = `<span class="material-symbols-outlined text-sm">school</span> Lehrer Modus`;
                viewToggle.classList.replace('bg-surface-container-high', 'bg-primary');
                viewToggle.classList.replace('text-on-surface', 'text-on-primary');
            }
        }

        function startSession() {
            if (studentSelect.value) {
                onboarding.classList.add('hidden');
                studentDashboard.classList.remove('hidden');
            } else {
                alert("Bitte wähle einen Namen aus!");
            }
        }

        function confirmTasks() {
            confirmationCard.classList.replace('bg-surface-container-lowest', 'bg-primary');
            confirmationCard.classList.replace('border-dashed', 'border-solid');
            confirmationCard.innerHTML = `
                <span class="material-symbols-outlined text-5xl text-on-primary" style="font-variation-settings: 'FILL' 1;">verified</span>
                <div class="space-y-2 text-on-primary">
                    <h3 class="text-h2 font-h2">Bestätigt!</h3>
                    <p class="text-body-md opacity-90">Die Liste wurde für heute gesperrt. Viel Erfolg beim Lernen!</p>
                </div>
                <div class="text-label-caps font-label-caps text-on-primary/70 uppercase">Sperre aktiv</div>
            `;
            // Lock all checkboxes
            document.querySelectorAll('#taskListContainer input').forEach(cb => cb.disabled = true);
        }
